# ResolveX – Environment Variables for Production
# ================================================
# Backend (Render) – add these in Render Dashboard → Environment
# Frontend (Vercel) – add VITE_API_URL in Vercel Dashboard → Settings → Environment Variables

# ----- BACKEND (Render) -----
NODE_ENV=production
PORT=10000
MONGODB_URI=                        # paste your Atlas connection string
JWT_ACCESS_SECRET=                  # 64-char random hex
JWT_REFRESH_SECRET=                 # 64-char random hex (different from above)
JWT_ACCESS_EXPIRES_IN=15m
JWT_REFRESH_EXPIRES_IN=7d
CLIENT_URL=                         # your Vercel URL, e.g. https://resolvex.vercel.app

# ----- FRONTEND (Vercel) -----
VITE_API_URL=                       # your Render backend URL, e.g. https://resolvex-backend.onrender.com
